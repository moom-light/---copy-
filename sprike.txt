extends Area2D

# 每次伤害数值
@export var damage: int = 1

# 同一个目标两次受伤之间的最小间隔（秒）
@export var damage_interval: float = 0.8

@onready var cooldown: Timer = $DamageCooldown

# 记录每个目标上次受伤的时间
var _last_damage_time: Dictionary = {}

func _ready() -> void:
	# 玩家进入地刺范围
	body_entered.connect(_on_body_entered)
	body_exited.connect(_on_body_exited)
	
	cooldown.one_shot = true
	cooldown.wait_time = damage_interval

# 玩家踩上地刺
func _on_body_entered(body: Node2D) -> void:
	_try_damage(body)

# 玩家离开地刺范围
func _on_body_exited(body: Node2D) -> void:
	if _last_damage_time.has(body):
		_last_damage_time.erase(body)

# 持续检测：玩家站在地刺上时循环掉血
func _physics_process(_delta: float) -> void:
	# 每帧检查当前与地刺重叠的所有 body
	for body in get_overlapping_bodies():
		_try_damage(body)

func _try_damage(body: Node2D) -> void:
	if not body.has_method("take_damage"):
		return
	
	# 检查冷却
	var now := Time.get_ticks_msec() / 1000.0
	var last: float = _last_damage_time.get(body, -999.0)
	if now - last < damage_interval:
		return
	
	_last_damage_time[body] = now
	body.take_damage(damage)
	# 如果是 CharacterBody2D，给一个向上的弹力
	if body is CharacterBody2D:
		body.velocity.y = -250.0
